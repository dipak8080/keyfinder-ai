# ---------- SPEECH TO TEXT (faster-whisper) ----------
# Pinned like everything else in this file. Two reasons beyond general
# reproducibility:
#
#   1. speech_to_text.py reads faster_whisper.tokenizer's PRIVATE
#      _LANGUAGE_CODES to build the language list served by
#      GET /speech-to-text/languages. That import is wrapped in a
#      try/except so an upstream rename degrades gracefully rather than
#      breaking the module - but it degrades to a hardcoded 35-code
#      subset, silently dropping ~64 languages from the UI. Pinning is
#      what stops that happening on an unrelated deploy.
#
#   2. The deploy workflow runs `docker builder prune -af` before every
#      build, so there is no layer cache and pip re-resolves from
#      scratch each time. Unpinned, two deploys of byte-identical source
#      could produce materially different images.
faster-whisper==1.2.1

# dependency-resolution error at build time, where it would be obvious.

# ---------- AUDIO TO SHEET MUSIC (/audio-to-sheet) ----------
# The sheet/ package's pipeline: transcription -> notation -> engrave.
# Modest footprint (~60-80MB total, mostly music21), nothing like the
# torch/essentia layers.
#
#   music21     MIDI -> MusicXML: quantise to the detected tempo grid,
#               insert key/time-sig, split piano into two staves.
#   verovio     MusicXML -> SVG/PDF engraving. Bundles its own music
#               fonts in the wheel, so no font package is needed for the
#               NOTATION itself.
#   cairosvg    per-page SVG -> PDF. Uses the libcairo2 already present on
#               this image via libcairo2-dev (installed for bgutil's
#               canvas build) - NO new system dependency.
#   pypdf       merge the per-page PDFs into one document.
#   pretty_midi MIDI parsing + note-count summary in the sheet pipeline.
#
# pretty_midi is left with a >= floor rather than a hard pin: it is a
# small pure-Python package and a hard == here risks a resolution
# conflict on a no-cache build if anything else ever constrains it.
music21==9.3.0
verovio==5.6.0
cairosvg==2.7.1
pypdf==6.1.1
pretty_midi>=0.2.10

# ---------- NOT INSTALLED: yt-dlp browser impersonation ----------
# Deliberately left out for now, documented so the gap is a decision
# rather than an oversight.
#
# `yt-dlp --list-impersonate-targets` currently reports every target
# (Chrome, Edge, Safari, Firefox, Tor) as "unavailable", because
# curl-cffi - which supplies the TLS/JA3 impersonation - is not
# installed. Enabling it is one line:
#
#     yt-dlp[default,curl-cffi]==2026.07.04
#
# Why this is not urgent for YouTube: yt-dlp has a client ladder there
# (android / ios / tv / web), so a blocked client falls through to
# another. Impersonation is a nice-to-have on top of that.
#
# Why it matters for TikTok: there is no equivalent ladder. Extraction
# currently works because TikTok accepts plain requests - the day it
# tightens JA3 fingerprinting, there is nothing to fall back to and the
# TikTok path breaks outright.
#
# Kept as a SEPARATE change from the transcription work on purpose:
# curl-cffi ships compiled binaries and changes yt-dlp's request handler,
# so if it destabilises extraction the cause should be unambiguous rather
# than tangled up with a Whisper deploy.
